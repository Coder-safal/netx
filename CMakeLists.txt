cmake_minimum_required(VERSION 3.20)
project(netx VERSION 0.1.0 DESCRIPTION "Linux network diagnostics" LANGUAGES CXX)

include(CTest)
include(GNUInstallDirs)

configure_file(cmake/version.hpp.in generated/netx/version.hpp @ONLY)

find_package(CURL REQUIRED)
add_library(netx_cli src/cli.cpp src/dns.cpp src/report.cpp src/web.cpp src/system.cpp)
target_link_libraries(netx_cli PUBLIC resolv CURL::libcurl)
target_compile_features(netx_cli PUBLIC cxx_std_17)
set_target_properties(netx_cli PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(netx_cli PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(netx_cli PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(netx src/main.cpp)
target_link_libraries(netx PRIVATE netx_cli)
install(TARGETS netx RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES docs/netx.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES completions/netx.bash DESTINATION ${CMAKE_INSTALL_DATADIR}/bash-completion/completions RENAME netx)
install(FILES docs/json-schema-1.json CLI_SPEC.md README.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})

set(CPACK_PACKAGE_NAME netx)
set(CPACK_PACKAGE_CONTACT "netx contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Linux network diagnostic CLI")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME offline_integration COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/checks.py $<TARGET_FILE:netx>)
    set_tests_properties(offline_integration PROPERTIES TIMEOUT 90)
    add_executable(netx_cli_tests tests/cli_test.cpp)
    target_link_libraries(netx_cli_tests PRIVATE netx_cli)
    add_test(NAME cli_unit COMMAND netx_cli_tests)
    add_executable(netx_dns_tests tests/dns_test.cpp)
    target_link_libraries(netx_dns_tests PRIVATE netx_cli)
    add_test(NAME dns_unit COMMAND netx_dns_tests)
    add_test(NAME cli_version COMMAND netx --version)
    set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "netx ${PROJECT_VERSION}")
    add_test(NAME cli_help COMMAND netx --help)
    set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "Usage: netx")
endif()
